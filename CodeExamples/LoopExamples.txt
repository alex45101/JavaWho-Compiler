println("Normal while loop counter:");

Int a;
a = 0;

while(a < 5)
{
    println(a);
    a = a + 1;
}

println("Do while loop counter:");

Int b;
b = 4;

while(true)
{
    println(b);

    b = b - 1;

    if(b < 0)
    {
        break;
    }
}

println("String loop counter with a's:");

String text;
text = "";

while (text != "aaaa")
{
    if(text == "aa")
    {
        println("double aa, yaya");
    }

    println(text);
    text = text + "a";
}
